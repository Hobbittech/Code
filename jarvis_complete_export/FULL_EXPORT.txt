async def fetch_weather(location: str) -> Optional[str]:
    try:
        async with httpx.AsyncClient(timeout=10.0) as client:
            geo = await client.get(
                "https://geocoding-api.open-meteo.com/v1/search",
                params={"name": location, "count": 1, "language": "en", "format": "json"},
            )
            geo_data = geo.json()
            if not geo_data.get("results"):
                return f"I could not locate '{location}', sir."
            place = geo_data["results"][0]
            lat, lon = place["latitude"], place["longitude"]
            display_name = f"{place['name']}, {place.get('country', '')}".strip(", ")

            wx = await client.get(
                "https://api.open-meteo.com/v1/forecast",
                params={
                    "latitude": lat, "longitude": lon,
                    "current": "temperature_2m,relative_humidity_2m,apparent_temperature,"
                               "is_day,precipitation,weather_code,wind_speed_10m",
                    "timezone": "auto",
                },
            )
            wd = wx.json().get("current", {})
            desc = WEATHER_CODE_MAP.get(wd.get("weather_code"), "conditions undetermined")
            return (
                f"Current weather in {display_name}: {desc}, "
                f"{wd.get('temperature_2m')}°C (feels like {wd.get('apparent_temperature')}°C), "
                f"humidity {wd.get('relative_humidity_2m')}%, "
                f"wind {wd.get('wind_speed_10m')} km/h."
            )
    except Exception as e:
        logger.exception("weather fetch failed")
        return f"Apologies, I could not retrieve the weather data ({e})."


async def web_search(query: str) -> Optional[str]:
    try:
        async with httpx.AsyncClient(timeout=10.0) as client:
            r = await client.get(
                "https://api.duckduckgo.com/",
                params={"q": query, "format": "json", "no_html": 1, "skip_disambig": 1},
            )
            data = r.json()
            parts = []
            if data.get("AbstractText"):
                parts.append(data["AbstractText"])
            elif data.get("Answer"):
                parts.append(data["Answer"])
            for rel in (data.get("RelatedTopics") or [])[:3]:
                if isinstance(rel, dict) and rel.get("Text"):
                    parts.append(rel["Text"])
            if not parts:
                return None
            return " | ".join(parts)[:1200]
    except Exception:
        logger.exception("web search failed")
        return None


TIME_TRIGGERS = re.compile(r"\b(time|date|day|today|tonight|what day|clock)\b", re.I)
WEATHER_TRIGGERS = re.compile(r"\b(weather|temperature|forecast|rain|snow|hot|cold|humid)\b", re.I)
SEARCH_TRIGGERS = re.compile(r"\b(search|look up|google|who is|what is|tell me about|news about|find)\b", re.I)


def extract_location(text: str) -> Optional[str]:
    m = re.search(r"\bin\s+([A-Za-z][A-Za-z\s,.'-]{1,40})", text)
    if m:
        loc = m.group(1).strip(" ?.,!").split("?")[0].split(".")[0]
        if loc:
            return loc.strip()
    return None


async def build_context(user_text: str) -> str:
    blocks: List[str] = []
    if TIME_TRIGGERS.search(user_text):
        blocks.append(f"TIME_CONTEXT: {get_time_context()}")
    if WEATHER_TRIGGERS.search(user_text):
        loc = extract_location(user_text) or "London"
        wx = await fetch_weather(loc)
        if wx:
            blocks.append(f"WEATHER_CONTEXT: {wx}")
    if SEARCH_TRIGGERS.search(user_text) and len(user_text) > 10:
        q = re.sub(r"^(please\s+)?(search|look up|google|tell me about|find)\s+", "", user_text, flags=re.I)
        result = await web_search(q.strip(" ?."))
        if result:
            blocks.append(f"SEARCH_CONTEXT: {result}")
    if not blocks:
        return user_text
    return "\n".join(blocks) + "\n\nUSER: " + user_text


class ChatRequest(BaseModel):
    session_id: str
    text: str
    speak: bool = True


class ChatResponse(BaseModel):
    session_id: str
    user_text: str
    reply_text: str
    audio_base64: Optional[str] = None
    timestamp: str


class Message(BaseModel):
    id: str = Field(default_factory=lambda: str(uuid.uuid4()))
    session_id: str
    role: str
    text: str
    timestamp: str = Field(default_factory=lambda: datetime.now(timezone.utc).isoformat())


async def store_message(session_id: str, role: str, text: str) -> None:
    msg = Message(session_id=session_id, role=role, text=text)
    await db.jarvis_messages.insert_one(msg.model_dump())


async def synthesize_voice(text: str) -> Optional[str]:
    try:
        snippet = text[:4000]
        return await tts_client.generate_speech_base64(text=snippet, model=TTS_MODEL, voice=TTS_VOICE)
    except Exception:
        logger.exception("tts failed")
        return None


async def run_chat(session_id: str, user_text: str, speak: bool) -> ChatResponse:
    chat = get_chat(session_id)
    augmented = await build_context(user_text)
    reply = await chat.send_message(UserMessage(text=augmented))
    reply_text = reply if isinstance(reply, str) else getattr(reply, "text", str(reply))

    await store_message(session_id, "user", user_text)
    await store_message(session_id, "assistant", reply_text)

    audio_b64 = await synthesize_voice(reply_text) if speak else None
    return ChatResponse(
        session_id=session_id,
        user_text=user_text,
        reply_text=reply_text,
        audio_base64=audio_b64,
        timestamp=datetime.now(timezone.utc).isoformat(),
    )


@api_router.get("/")
async def root():
    return {"system": "J.A.R.V.I.S.", "status": "online", "model": CHAT_MODEL_NAME}


@api_router.get("/health")
async def health():
    return {"status": "operational", "time": get_time_context()}


@api_router.post("/chat", response_model=ChatResponse)
async def chat_endpoint(req: ChatRequest):
    if not req.text.strip():
        raise HTTPException(status_code=400, detail="text is required")
    return await run_chat(req.session_id, req.text.strip(), req.speak)


@api_router.post("/voice", response_model=ChatResponse)
async def voice_endpoint(
    session_id: str = Form(...),
    audio: UploadFile = File(...),
    speak: bool = Form(True),
):
    audio_bytes = await audio.read()
    if not audio_bytes:
        raise HTTPException(status_code=400, detail="empty audio file")
    buf = io.BytesIO(audio_bytes)
    buf.name = audio.filename or "audio.webm"
    try:
        transcript = await stt_client.transcribe(file=buf, model="whisper-1", response_format="json")
        user_text = (transcript.text or "").strip()
    except Exception as e:
        logger.exception("whisper transcription failed")
        raise HTTPException(status_code=500, detail=f"transcription failed: {e}")
    if not user_text:
        raise HTTPException(status_code=400, detail="could not transcribe audio")
    return await run_chat(session_id, user_text, speak)


@api_router.get("/history/{session_id}")
async def history(session_id: str):
    docs = (
        await db.jarvis_messages.find({"session_id": session_id}, {"_id": 0})
        .sort("timestamp", 1)
        .to_list(500)
    )
    return {"session_id": session_id, "messages": docs}


@api_router.delete("/history/{session_id}")
async def clear_history(session_id: str):
    await db.jarvis_messages.delete_many({"session_id": session_id})
    _chat_sessions.pop(session_id, None)
    return {"status": "cleared", "session_id": session_id}

function App() {
    const [sessionId] = useState(getSessionId);
    const [messages, setMessages] = useState([]);
    const [input, setInput] = useState("");
    const [mode, setMode] = useState("idle");
    const [muted, setMuted] = useState(false);
    const [status, setStatus] = useState("online");

    const audioRef = useRef(null);
    const mediaRecorderRef = useRef(null);
    const audioChunksRef = useRef([]);
    const scrollRef = useRef(null);
    const recordingRef = useRef(false);

    useEffect(() => {
        axios.get(`${API}/health`).then(() => setStatus("online")).catch(() => setStatus("offline"));
        axios.get(`${API}/history/${sessionId}`).then((r) => {
            if (r.data?.messages?.length) setMessages(r.data.messages.map((m) => ({ role: m.role, text: m.text })));
        }).catch(() => {});
    }, [sessionId]);

    useEffect(() => {
        if (scrollRef.current) scrollRef.current.scrollTop = scrollRef.current.scrollHeight;
    }, [messages, mode]);

    const playAudio = useCallback((base64) => {
        if (!base64 || muted) return;
        try {
            const src = `data:audio/mp3;base64,${base64}`;
            if (!audioRef.current) audioRef.current = new Audio();
            audioRef.current.src = src;
            audioRef.current.onplay = () => setMode("speaking");
            audioRef.current.onended = () => setMode("idle");
            audioRef.current.onerror = () => setMode("idle");
            audioRef.current.play().catch(() => setMode("idle"));
        } catch { setMode("idle"); }
    }, [muted]);

    const sendText = useCallback(async (text) => {
        if (!text.trim() || mode === "thinking") return;
        setMessages((m) => [...m, { role: "user", text }]);
        setInput("");
        setMode("thinking");
        try {
            const r = await axios.post(`${API}/chat`, { session_id: sessionId, text, speak: !muted });
            setMessages((m) => [...m, { role: "assistant", text: r.data.reply_text }]);
            if (r.data.audio_base64 && !muted) playAudio(r.data.audio_base64);
            else setMode("idle");
        } catch (e) {
            setMessages((m) => [...m, { role: "assistant", text: "Apologies — communication interrupted." }]);
            setMode("idle");
        }
    }, [mode, sessionId, muted, playAudio]);

    const startRecording = useCallback(async () => {
        if (recordingRef.current || mode === "thinking") return;
        try {
            const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
            audioChunksRef.current = [];
            const mr = new MediaRecorder(stream, { mimeType: "audio/webm" });
            mr.ondataavailable = (e) => { if (e.data.size > 0) audioChunksRef.current.push(e.data); };
            mr.onstop = async () => {
                stream.getTracks().forEach((t) => t.stop());
                const blob = new Blob(audioChunksRef.current, { type: "audio/webm" });
                if (blob.size < 1000) { setMode("idle"); return; }
                setMode("thinking");
                const fd = new FormData();
                fd.append("session_id", sessionId);
                fd.append("audio", blob, "speech.webm");
                fd.append("speak", muted ? "false" : "true");
                try {
                    const r = await axios.post(`${API}/voice`, fd, { headers: { "Content-Type": "multipart/form-data" } });
                    setMessages((m) => [...m,
                        { role: "user", text: r.data.user_text },
                        { role: "assistant", text: r.data.reply_text }]);
                    if (r.data.audio_base64 && !muted) playAudio(r.data.audio_base64);
                    else setMode("idle");
                } catch {
                    setMessages((m) => [...m, { role: "assistant", text: "I'm afraid I couldn't quite catch that, sir." }]);
                    setMode("idle");
                }
            };
            mr.start();
            mediaRecorderRef.current = mr;
            recordingRef.current = true;
            setMode("listening");
        } catch {
            alert("Microphone permission denied or unavailable.");
            setMode("idle");
        }
    }, [mode, sessionId, muted, playAudio]);

    const stopRecording = useCallback(() => {
        if (!recordingRef.current) return;
        recordingRef.current = false;
        try { mediaRecorderRef.current?.stop(); } catch {}
    }, []);

    const toggleRecord = useCallback(() => {
        if (recordingRef.current) stopRecording(); else startRecording();
    }, [startRecording, stopRecording]);

    const clearHistory = useCallback(async () => {
        if (!window.confirm("Clear conversation history?")) return;
        try { await axios.delete(`${API}/history/${sessionId}`); } catch {}
        setMessages([]);
    }, [sessionId]);

    return (
        <div className="App relative hud-grid">
            <div className="hud-vignette" />
            <div className="hud-noise" />
            <div className="hud-scanline" />
            <div className="relative z-10 h-screen flex flex-col">
                <HudHeader status={status} />
                <div className="flex-1 grid grid-cols-12 gap-6 px-8 pb-6" style={{ minHeight: 0 }}>
                    <aside className="col-span-3 flex flex-col">
                        <Telemetry />
                        <QuickPrompts onPick={(p) => sendText(p)} />
                    </aside>
                    <section className="col-span-5 flex flex-col items-center justify-between">
                        <div className="flex-1 flex flex-col items-center justify-center w-full">
                            <Orb mode={mode} />
                            <StatusReadout mode={mode} />
                        </div>
                        <div className="w-full flex flex-col items-center gap-4 mt-6">
                            <div className="flex items-center gap-6">
                                <button onClick={() => setMuted((v) => !v)} className="btn-circle" style={{ width: 48, height: 48 }} title={muted ? "Unmute" : "Mute"}>
                                    {muted ? <VolumeX size={20} /> : <Volume2 size={20} />}
                                </button>
                                <button onClick={toggleRecord} className={`btn-circle ${mode === "listening" ? "recording" : ""}`} title="Tap to speak">
                                    {mode === "listening" ? <MicOff size={28} /> : <Mic size={28} />}
                                </button>
                                <button onClick={clearHistory} className="btn-circle" style={{ width: 48, height: 48 }} title="Clear history">
                                    <Trash2 size={18} />
                                </button>
                            </div>
                            <form onSubmit={(e) => { e.preventDefault(); sendText(input); }} className="w-full flex gap-2 mt-2">
                                <input className="input-hud" placeholder="Type a directive, sir…" value={input} onChange={(e) => setInput(e.target.value)} disabled={mode === "thinking"} />
                                <button type="submit" className="btn-hud btn-hud-primary flex items-center gap-2" disabled={mode === "thinking" || !input.trim()}>
                                    {mode === "thinking" ? <Loader2 size={14} className="animate-spin" /> : <Send size={14} />}
                                    TRANSMIT
                                </button>
                            </form>
                        </div>
                    </section>
                    <aside className="col-span-4 flex flex-col" style={{ minHeight: 0 }}>
                        <Conversation messages={messages} thinking={mode === "thinking"} scrollRef={scrollRef} />
                    </aside>
                </div>
                <footer className="px-8 py-3 border-t border-[rgba(0,229,255,0.2)] flex items-center justify-between text-[10px] tracking-[0.3em] hud-text-ash uppercase">
                    <span>SESSION ID :: {sessionId.slice(0, 18)}…</span>
                    <span className="flex items-center gap-2"><Power size={10} /> ARC REACTOR :: NOMINAL</span>
                    <span>STARK INDUSTRIES // R&D</span>
                </footer>
            </div>
        </div>
    );
}
